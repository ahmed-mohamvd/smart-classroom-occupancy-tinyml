void UART_Print(const char *message)
{
  HAL_UART_Transmit(&huart2,
                    (uint8_t *)message,
                    strlen(message),
                    HAL_MAX_DELAY);
}

void TOF_PrintApiResult(const char *step, VL53L0X_Error status)
{
  snprintf(uart_string,
           sizeof(uart_string),
           "%s: API status=%d\r\n",
           step,
           (int)status);
  UART_Print(uart_string);
}

VL53L0X_Error TOF_Init(void)
{
  VL53L0X_Error status;

  if (HAL_I2C_IsDeviceReady(&hi2c1,
                            VL53L0X_ADDRESS_HAL,
                            3,
                            100) != HAL_OK)
  {
    UART_Print("VL53L0X did not answer at 0x29.\r\n");
    return VL53L0X_ERROR_CONTROL_INTERFACE;
  }

  memset(&tof_device, 0, sizeof(tof_device));
  tof_device.I2cDevAddr = VL53L0X_ADDRESS_HAL;

  status = VL53L0X_DataInit(&tof_device);
  TOF_PrintApiResult("VL53L0X_DataInit", status);
  if (status != VL53L0X_ERROR_NONE)
  {
    return status;
  }

  status = VL53L0X_StaticInit(&tof_device);
  TOF_PrintApiResult("VL53L0X_StaticInit", status);
  if (status != VL53L0X_ERROR_NONE)
  {
    return status;
  }

  status = VL53L0X_PerformRefSpadManagement(&tof_device,
                                             &tof_ref_spad_count,
                                             &tof_is_aperture_spads);
  TOF_PrintApiResult("VL53L0X_RefSpad", status);
  if (status != VL53L0X_ERROR_NONE)
  {
    return status;
  }

  status = VL53L0X_PerformRefCalibration(&tof_device,
                                          &tof_vhv_settings,
                                          &tof_phase_cal);
  TOF_PrintApiResult("VL53L0X_RefCalibration", status);
  if (status != VL53L0X_ERROR_NONE)
  {
    return status;
  }

  status = VL53L0X_SetDeviceMode(&tof_device,
                                  VL53L0X_DEVICEMODE_SINGLE_RANGING);
  TOF_PrintApiResult("VL53L0X_SetDeviceMode", status);
  if (status != VL53L0X_ERROR_NONE)
  {
    return status;
  }

  /* ST high-accuracy profile: stricter quality limits and 200 ms budget. */
  status = VL53L0X_SetLimitCheckValue(
      &tof_device,
      VL53L0X_CHECKENABLE_SIGNAL_RATE_FINAL_RANGE,
      (FixPoint1616_t)16384U);   /* 0.25 MCps in 16.16 format */
  TOF_PrintApiResult("VL53L0X_SetSignalLimit", status);
  if (status != VL53L0X_ERROR_NONE)
  {
    return status;
  }

  status = VL53L0X_SetLimitCheckValue(
      &tof_device,
      VL53L0X_CHECKENABLE_SIGMA_FINAL_RANGE,
      (FixPoint1616_t)1179648U); /* 18 mm in 16.16 format */
  TOF_PrintApiResult("VL53L0X_SetSigmaLimit", status);
  if (status != VL53L0X_ERROR_NONE)
  {
    return status;
  }

  status = VL53L0X_SetMeasurementTimingBudgetMicroSeconds(&tof_device,
                                                           200000U);
  TOF_PrintApiResult("VL53L0X_SetTimingBudget", status);
  return status;
}

/**
  * @brief  The application entry point.
  * @retval int
  */
int main(void)
{
  /* USER CODE BEGIN 1 */

  /* USER CODE END 1 */

  HAL_Init();
  SystemClock_Config();

  MX_GPIO_Init();
  MX_USART2_UART_Init();
  MX_I2C1_Init();

  /* USER CODE BEGIN 2 */
  UART_Print("=== VL53L0X + OLED DISTANCE TEST ===\r\n");

  SSD1306_Init();
  SSD1306_GotoXY(0, 0);
  SSD1306_Puts("VL53L0X ToF", &Font_7x10, 1);
  SSD1306_GotoXY(0, 22);
  SSD1306_Puts("Initializing...", &Font_7x10, 1);
  SSD1306_UpdateScreen();

  tof_status = TOF_Init();
  tof_initialized = (tof_status == VL53L0X_ERROR_NONE) ? 1U : 0U;

  if (tof_initialized != 0U)
  {
    UART_Print("VL53L0X initialization OK.\r\n");
  }
  else
  {
    snprintf(uart_string,
             sizeof(uart_string),
             "VL53L0X initialization FAILED: API status=%d\r\n",
             (int)tof_status);
    UART_Print(uart_string);

    SSD1306_GotoXY(0, 22);
    SSD1306_Puts("Init FAILED    ", &Font_7x10, 1);
    SSD1306_GotoXY(0, 44);
    snprintf(oled_string,
             sizeof(oled_string),
             "API error:%d    ",
             (int)tof_status);
    SSD1306_Puts(oled_string, &Font_7x10, 1);
    SSD1306_UpdateScreen();
  }
  /* USER CODE END 2 */

  /* Infinite loop */
  /* USER CODE BEGIN WHILE */
  while (1)
  {
    if (tof_initialized != 0U)
    {
      uint32_t signal_x1000;
      uint32_t ambient_x1000;

      tof_status = VL53L0X_PerformSingleRangingMeasurement(&tof_device,
                                                            &tof_measurement);

      if (tof_status == VL53L0X_ERROR_NONE)
      {
        signal_x1000 = (uint32_t)(((uint64_t)tof_measurement.SignalRateRtnMegaCps *
                                   1000ULL) >> 16);
        ambient_x1000 = (uint32_t)(((uint64_t)tof_measurement.AmbientRateRtnMegaCps *
                                    1000ULL) >> 16);

        snprintf(uart_string,
                 sizeof(uart_string),
                 "Distance=%u mm, Status=%u, Signal=%lu.%03lu, Ambient=%lu.%03lu MCps\r\n",
                 (unsigned int)tof_measurement.RangeMilliMeter,
                 (unsigned int)tof_measurement.RangeStatus,
                 (unsigned long)(signal_x1000 / 1000U),
                 (unsigned long)(signal_x1000 % 1000U),
                 (unsigned long)(ambient_x1000 / 1000U),
                 (unsigned long)(ambient_x1000 % 1000U));
        UART_Print(uart_string);

        SSD1306_GotoXY(0, 0);
        SSD1306_Puts("VL53L0X ToF    ", &Font_7x10, 1);

        if (tof_measurement.RangeStatus == 0U)
        {
          snprintf(oled_string,
                   sizeof(oled_string),
                   "Dist:%4u mm    ",
                   (unsigned int)tof_measurement.RangeMilliMeter);
          SSD1306_GotoXY(0, 22);
          SSD1306_Puts(oled_string, &Font_7x10, 1);
          SSD1306_GotoXY(0, 44);
          SSD1306_Puts("Measurement: OK", &Font_7x10, 1);
        }
        else
        {
          SSD1306_GotoXY(0, 22);
          SSD1306_Puts("No valid target ", &Font_7x10, 1);
          SSD1306_GotoXY(0, 44);
          snprintf(oled_string,
                   sizeof(oled_string),
                   "Range status:%u ",
                   (unsigned int)tof_measurement.RangeStatus);
          SSD1306_Puts(oled_string, &Font_7x10, 1);
        }

        SSD1306_UpdateScreen();
      }
      else
      {
        snprintf(uart_string,
                 sizeof(uart_string),
                 "VL53L0X ranging API error=%d\r\n",
                 (int)tof_status);
        UART_Print(uart_string);

        SSD1306_GotoXY(0, 22);
        SSD1306_Puts("Ranging FAILED  ", &Font_7x10, 1);
        SSD1306_UpdateScreen();
      }
    }

    HAL_Delay(250);

    /* USER CODE END WHILE */
    /* USER CODE BEGIN 3 */
  }
  /* USER CODE END 3 */
}

/**
  * @brief System Clock Configuration
  * @retval None
  */
void SystemClock_Config(void)
{
  RCC_OscInitTypeDef RCC_OscInitStruct = {0};
  RCC_ClkInitTypeDef RCC_ClkInitStruct = {0};

  if (HAL_PWREx_ControlVoltageScaling(PWR_REGULATOR_VOLTAGE_SCALE1) != HAL_OK)
  {
    Error_Handler();
  }

  RCC_OscInitStruct.OscillatorType = RCC_OSCILLATORTYPE_HSI;
  RCC_OscInitStruct.HSIState = RCC_HSI_ON;
  RCC_OscInitStruct.HSICalibrationValue = RCC_HSICALIBRATION_DEFAULT;
  RCC_OscInitStruct.PLL.PLLState = RCC_PLL_ON;
  RCC_OscInitStruct.PLL.PLLSource = RCC_PLLSOURCE_HSI;
  RCC_OscInitStruct.PLL.PLLM = 1;
  RCC_OscInitStruct.PLL.PLLN = 10;
  RCC_OscInitStruct.PLL.PLLP = RCC_PLLP_DIV7;
  RCC_OscInitStruct.PLL.PLLQ = RCC_PLLQ_DIV2;
  RCC_OscInitStruct.PLL.PLLR = RCC_PLLR_DIV2;

  if (HAL_RCC_OscConfig(&RCC_OscInitStruct) != HAL_OK)
  {
    Error_Handler();
  }

  RCC_ClkInitStruct.ClockType = RCC_CLOCKTYPE_HCLK | RCC_CLOCKTYPE_SYSCLK |
                              RCC_CLOCKTYPE_PCLK1 | RCC_CLOCKTYPE_PCLK2;
  RCC_ClkInitStruct.SYSCLKSource = RCC_SYSCLKSOURCE_PLLCLK;
  RCC_ClkInitStruct.AHBCLKDivider = RCC_SYSCLK_DIV1;
  RCC_ClkInitStruct.APB1CLKDivider = RCC_HCLK_DIV1;
  RCC_ClkInitStruct.APB2CLKDivider = RCC_HCLK_DIV1;

  if (HAL_RCC_ClockConfig(&RCC_ClkInitStruct, FLASH_LATENCY_4) != HAL_OK)
  {
    Error_Handler();
  }
}
