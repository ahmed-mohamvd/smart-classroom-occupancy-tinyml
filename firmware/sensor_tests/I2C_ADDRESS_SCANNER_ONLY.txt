void UART_Print(const char *message)
{
  HAL_UART_Transmit(&huart2,
                    (uint8_t *)message,
                    strlen(message),
                    HAL_MAX_DELAY);
}

const char *I2C_DeviceName(uint8_t address)
{
  switch (address)
  {
    case 0x29:
      return "VL53L0X ToF distance sensor";

    case 0x23:
      return "BH1750 (ADDR = GND)";

    case 0x5C:
      return "BH1750 (ADDR = 3.3V)";

    case 0x3C:
    case 0x3D:
      return "SSD1306 OLED";

    case 0x76:
    case 0x77:
      return "BME280";

    default:
      return "Unknown I2C device";
  }
}

void I2C1_ScanAllAddresses(void)
{
  uint8_t address;
  uint8_t found_count = 0;
  HAL_StatusTypeDef status;

  UART_Print("\r\n========================================\r\n");
  UART_Print("Scanning every valid I2C1 address...\r\n");
  UART_Print("7-bit address range: 0x08 to 0x77\r\n");

  for (address = 0x08U; address <= 0x77U; address++)
  {
    /* STM32 HAL requires the 7-bit address shifted left by one bit. */
    status = HAL_I2C_IsDeviceReady(&hi2c1,
                                   (uint16_t)(address << 1),
                                   3,
                                   20);

    if (status == HAL_OK)
    {
      found_count++;

      snprintf(uart_message,
               sizeof(uart_message),
               "FOUND: 7-bit=0x%02X, HAL=0x%02X -> %s\r\n",
               (unsigned int)address,
               (unsigned int)(address << 1),
               I2C_DeviceName(address));
      UART_Print(uart_message);
    }
  }

  if (found_count == 0U)
  {
    UART_Print("RESULT: No I2C device answered.\r\n");
    UART_Print("Check VCC, GND, SDA, SCL and the selected I2C peripheral.\r\n");
  }
  else
  {
    snprintf(uart_message,
             sizeof(uart_message),
             "RESULT: %u I2C device(s) found.\r\n",
             (unsigned int)found_count);
    UART_Print(uart_message);
  }

  UART_Print("Expected: VL53L0X=0x29, BH1750=0x23/0x5C.\r\n");
  UART_Print("========================================\r\n");
}

/**
  * @brief  The application entry point.
  * @retval int
  */
int main(void)
{
  /* USER CODE BEGIN 1 */

  /* USER CODE END 1 */

  HAL_Init();
  SystemClock_Config();

  MX_GPIO_Init();
  MX_USART2_UART_Init();
  MX_I2C1_Init();

  /* USER CODE BEGIN 2 */
  UART_Print("\r\n=== STM32 I2C1 ADDRESS SCANNER STARTED ===\r\n");
  UART_Print("OLED code is not used in this test.\r\n");
  /* USER CODE END 2 */

  /* Infinite loop */
  /* USER CODE BEGIN WHILE */
  while (1)
  {
    I2C1_ScanAllAddresses();

    /* Repeat the complete scan every 3 seconds. */
    HAL_Delay(3000);

    /* USER CODE END WHILE */
    /* USER CODE BEGIN 3 */
  }
  /* USER CODE END 3 */
}

/**
  * @brief System Clock Configuration
  * @retval None
  */
void SystemClock_Config(void)
{
  RCC_OscInitTypeDef RCC_OscInitStruct = {0};
  RCC_ClkInitTypeDef RCC_ClkInitStruct = {0};

  if (HAL_PWREx_ControlVoltageScaling(PWR_REGULATOR_VOLTAGE_SCALE1) != HAL_OK)
  {
    Error_Handler();
  }

  RCC_OscInitStruct.OscillatorType = RCC_OSCILLATORTYPE_HSI;
  RCC_OscInitStruct.HSIState = RCC_HSI_ON;
  RCC_OscInitStruct.HSICalibrationValue = RCC_HSICALIBRATION_DEFAULT;
  RCC_OscInitStruct.PLL.PLLState = RCC_PLL_ON;
  RCC_OscInitStruct.PLL.PLLSource = RCC_PLLSOURCE_HSI;
  RCC_OscInitStruct.PLL.PLLM = 1;
  RCC_OscInitStruct.PLL.PLLN = 10;
  RCC_OscInitStruct.PLL.PLLP = RCC_PLLP_DIV7;
  RCC_OscInitStruct.PLL.PLLQ = RCC_PLLQ_DIV2;
  RCC_OscInitStruct.PLL.PLLR = RCC_PLLR_DIV2;

  if (HAL_RCC_OscConfig(&RCC_OscInitStruct) != HAL_OK)
  {
    Error_Handler();
  }

  RCC_ClkInitStruct.ClockType = RCC_CLOCKTYPE_HCLK | RCC_CLOCKTYPE_SYSCLK |
                              RCC_CLOCKTYPE_PCLK1 | RCC_CLOCKTYPE_PCLK2;
  RCC_ClkInitStruct.SYSCLKSource = RCC_SYSCLKSOURCE_PLLCLK;
  RCC_ClkInitStruct.AHBCLKDivider = RCC_SYSCLK_DIV1;
  RCC_ClkInitStruct.APB1CLKDivider = RCC_HCLK_DIV1;
  RCC_ClkInitStruct.APB2CLKDivider = RCC_HCLK_DIV1;

  if (HAL_RCC_ClockConfig(&RCC_ClkInitStruct, FLASH_LATENCY_4) != HAL_OK)
  {
    Error_Handler();
  }
}
